# win2blur v3.0 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Per-window effect profiles, focus differentiation with smooth transitions, and a mouse-following local blur circle, all on the crisp_overlay unified engine.

**Architecture:** crisp_overlay.exe becomes the unified blur engine (per-window process): one cached DDA background frame, re-blurred on parameter changes without re-capture (zero flicker for focus/mouse/circle changes), with internal focus tracking (×0.6 background reduction) and 300ms ease transitions. tray_app.exe orchestrates: [Apps] profile table (AutoFrost upgrade), Settings UI (profile editor + circle sliders + focus factor), and crash-guard restart of overlay processes.

**Tech Stack:** C++17, MinGW winlibs (g++ 13+), Win32, DDA (DXGI), CMake. No new dependencies. Test infra: a tiny assert-based unit target for pure logic + a zero-dependency Python smoke script (tkinter + ctypes). Spec: `docs/superpowers/specs/2026-08-07-win2blur-v3-design.md`.

## Global Constraints

- Windows 10 Pro 19045 only target; no Win11-only APIs
- No new dependencies (no external libs, no Python packages beyond stdlib)
- Existing build: `cmake -B build5 -S native && cmake --build build5`
- Child exes embedded as RCDATA: 101=acrylic_overlay, 102=welcome_demo, 103=crisp_overlay
- Overlay processes must remain independent (crash isolation); tray may only signal via process kill/relaunch and CLI args
- CLI output/strings in English (Windows cmd garbles Chinese)
- Deploy per memory [[win2blur-dist-autoupdate]]: build5 exes → `win2blur/dist/` + `dist_release_3.0/`
- Commit per [[git-daily-single-version]]: one commit per day, squash intraday to latest

  ```cpp
  struct ProfileEntry {
      std::wstring exe;      // process exe name, lowercase, e.g. "notepad.exe"
      std::wstring cls;      // optional window class filter, empty = any
      int mode;              // 0=acrylic, 1=crisp; -1 = use global
  ```

**Key behavior:** fields separated by spaces, `key=value`; exe token is everything before first `=`. Value tokens lowercased for comparison (`on/off/acrylic/crisp`). Malformed/empty line → `{exe:L""}`.

```cpp
// profile.cpp
#include "profile.h"
#include <vector>
static std::wstring lower(const std::wstring& s) {
    std::wstring r = s; for (auto& c : r) c = (wchar_t)towlower(c); return r;
}
static int parse_int_field(const std::wstring& v, int lo, int hi) {
    wchar_t* end = nullptr; long n = wcstol(v.c_str(), &end, 10);
    if (!end || *end != 0 || end == v.c_str()) return -1;
    return (n >= lo && n <= hi) ? (int)n : -1;
}
ProfileEntry parse_profile_line(const wchar_t* line) {
    ProfileEntry e; if (!line || !*line) return e;
    std::wstring s(line);
    size_t eq = s.find(L'=');
    std::wstring exeTok = eq == std::wstring::npos ? s : s.substr(0, eq);
    while (!exeTok.empty() && exeTok.back() == L' ') exeTok.pop_back();
    if (exeTok.empty()) return e;
    e.exe = lower(exeTok);
    if (eq == std::wstring::npos) return e;
    std::wstring rest = s.substr(eq + 1);
    size_t i = 0;
    while (i < rest.size()) {
        while (i < rest.size() && rest[i] == L' ') i++;
        size_t j = rest.find(L' ', i);
        if (j == std::wstring::npos) j = rest.size();
        std::wstring tok = rest.substr(i, j - i); i = j + 1;
        size_t keq = tok.find(L'=');
        if (keq == std::wstring::npos) continue;
        std::wstring k = lower(tok.substr(0, keq));
        std::wstring v = lower(tok.substr(keq + 1));
        if (k == L"mode") e.mode = (v == L"crisp") ? 1 : (v == L"acrylic") ? 0 : -1;
        else if (k == L"alpha") e.alpha = parse_int_field(v, 0, 255);
        else if (k == L"radius") e.radius = parse_int_field(v, 1, 120);
        else if (k == L"tint") e.tint = parse_int_field(v, 0, 100);
        else if (k == L"circle") e.circle = (v == L"on");
        else if (k == L"cls") e.cls = v;
    }
    return e;
}
```

```python
"""v3.0 smoke test: launch crisp_overlay against a tkinter window, assert lifecycle."""
import ctypes, subprocess, sys, time, os
import tkinter as tk
user32 = ctypes.windll.user32
CRISP_CLS = "CrispOverlayClass"
BUILD = os.path.join(os.path.dirname(__file__), "..", "native", "build5", "crisp_overlay.exe")

def find_overlay():
    hwnd = user32.FindWindowW(CRISP_CLS, None)
    return hwnd

def run_case(args, expect_alive=True, wait=1.5):
    target = tk.Tk(); target.geometry("300x200+100+100"); target.update()
    target_hwnd = ctypes.c_void_p(target.winfo_id())
    root_hwnd = target.winfo_id()  # toplevel of tk
    hwnd = user32.GetParent(ctypes.c_void_p(root_hwnd)) or root_hwnd
    proc = subprocess.Popen([BUILD, "0x%08X" % hwnd] + args, stdout=subprocess.PIPE, stderr=subprocess.PIPE)
    time.sleep(wait)
    overlay = find_overlay()
    alive = proc.poll() is None
    ok = (alive == expect_alive)
    print(("PASS" if ok else "FAIL"), "case", args, "overlay=%s alive=%s" % (bool(overlay), alive))
    if overlay: user32.PostMessageW(overlay, 0x0010, 0, 0)  # WM_CLOSE
    if proc.poll() is None: proc.terminate()
    target.destroy()
    return ok

if __name__ == "__main__":
    results = [
        run_case(["120", "15"]),                           # back-compat 2-arg (wash=120, tint=15)
        run_case(["120", "20", "15", "50", "200", "30", "60"]),  # wash tint radius circle boost band focus
        run_case(["0x0", "120", "15"], expect_alive=False),     # invalid hwnd -> exit
    ]
    print("ALL PASS" if all(results) else "SOME FAILED")
    sys.exit(0 if all(results) else 1)
```

```cpp
    DWORD pid=0; GetWindowThreadProcessId(g_target,&pid);
    if(pid!=g_targetPid){ wprintf(L"[!] Target HWND recycled, exiting\n"); g_running=false; return; }

    ShowWindow(g_overlay,SW_HIDE); ShowWindow(g_target,SW_HIDE); DwmFlush();
    g_cache.clear();
    bool ok=capture_gpu(r.left-g_outX,r.top-g_outY,w,h,g_cache);
    ShowWindow(g_target,SW_SHOW); ShowWindow(g_overlay,SW_SHOWNOACTIVATE); reanchor();
    if(ok&&!g_cache.empty()){ g_cacheValid=true; g_cachedRect=r; g_renderDirty=1; render_from_cache(); }
    settleCount=0;
}
```

### Task 4: crisp_overlay — focus tracking + 300ms ease transition

**Files:**
- Modify: `native/src/crisp_overlay.cpp`

**Interfaces:**
- Consumes: existing `WinEventProc` foreground branch (already fires on ANY foreground change), poll tick
- Produces:
  ```cpp
  static bool g_focused = true;            // is g_target the foreground window
  static DWORD g_focusChangedAt = 0;       // tick count when focus state flipped
  static float g_curFocusF = 1.0f;         // current (lerped) focus factor
  // target focus factor; 1.0 when focused, g_focusPct/100 when not
  static float focus_target();             // { return g_focused ? 1.0f : g_focusPct / 100.0f; }
  static float ease_out_cubic(float t);    // { return 1 - pow(1 - t, 3); }
  ```

**Behavior:** Focused → all params at profile values. Unfocused → radius×F, wash×F, tint×0.5 (F = g_focusPct/100; tint factor independently eased as `g_tintF`, target 1.0 focused / 0.5 unfocused — see Task 4 fix round 1, commit 59dcae2; the ×5/6 scheme was revoked). On focus flip, record `g_focusChangedAt`; per frame, `g_curFocusF` lerps start→target with ease-out-cubic over 300ms. `render_from_cache()` uses `g_curFocusF` and marks itself dirty while `|g_curFocusF - target| > 0.001`.

- [ ] **Step 1: Add focus state + helpers** (as Interfaces above; `#include <cmath>` if not present).

- [ ] **Step 2: Wire foreground detection** — in `tick()`, before the moved check:

```cpp
    bool fg = GetForegroundWindow() == g_target;
    if (fg != g_focused) { g_focused = fg; g_focusChangedAt = GetTickCount(); }
    // ease toward target
    float t = 1.0f;
    if (g_focusChangedAt) {
        DWORD dt = GetTickCount() - g_focusChangedAt;
        t = dt >= 300 ? 1.0f : ease_out_cubic((float)dt / 300.0f);
    }
    float from = g_focusChangedAt ? g_curFocusF : focus_target();
    float to = focus_target();
    g_curFocusF = from + (to - from) * t;
    if (fabsf(g_curFocusF - to) > 0.001f) g_renderDirty = 1;
```
(When transition finishes, `g_focusChangedAt` may stay set; harmless since t clamps to 1.)

### Task 5: crisp_overlay — mouse-follow blur circle (dual blur + radial mask + idle fade)

**Files:**
- Modify: `native/src/crisp_overlay.cpp`

**Interfaces:**
- Consumes: `g_circlePx/g_boostX100/g_bandPx` (Task 2), cached frame (Task 3)
- Produces:
  ```cpp
  static POINT g_circleCenter = {-1,-1};   // cursor pos relative to target rect; -1 = hidden
  static DWORD g_lastMouseAt = 0;          // last cursor-move tick
  static float g_circleAlpha = 1.0f;       // 0..1, eased for idle fade
  // radial smoothstep mask value at distance d (0 outside circle+band)
  static float circle_mask(int d, int r, int band);
  ```
  `circle_mask`: `float x = (float)(d - r) / band; x = x<0?0:(x>1?1:x); return 1 - x*x*(3-2*x);`

**Behavior:** Each tick: `GetCursorPos`; if inside target rect → update `g_circleCenter` (relative), `g_lastMouseAt=GetTickCount()`, dirty. Idle >1500ms → `g_circleAlpha` eases toward 0 over 300ms (else toward 1), dirty while easing. render_from_cache(): when circle active, blur the cached frame twice — base at current radius, boost at `radius*boostX100/100` — and blend per pixel: `out = base*(1-m)*1... ` precisely: `out = base*(1-mask) + boost*mask`, then tint, then wash alpha via render_layered with per-pixel alpha from mask? No — wash is uniform via BLENDFUNCTION; the per-pixel blend already encodes the circle (both layers at full alpha, blended by mask, then uniform wash alpha over all). That gives "circle more blur, outside base" with uniform wash — correct per design.

**Performance:** blur at 1/4 scale: downscale g_cache → work buffer sw=w/4, sh=h/4, run both blurs there, blend, upscale to full, tint, render. Implement `void downscale(const std::vector<BYTE>& src, int w,int h, int sw,int sh, std::vector<BGRA>& out)` (box average of 4x4 groups) and `void upscale_blend(const std::vector<BGRA>& low, int sw,int sh, int w,int h, std::vector<BGRA>& out)` (nearest neighbor, applying circle mask with center scaled by 4).

- [ ] **Step 1: Add circle state + circle_mask** (as Interfaces). Include `<math.h>` for fabsf already used.

- [ ] **Step 2: Track cursor in tick()** — after focus wiring. FIXED in implementation (d4b274d) vs the original snippet: stamp `g_lastMouseAt` + dirty **only on actual cursor-position change** (the original stamped every tick while inside, which made idle never exceed the poll interval so the fade could never fire, and froze the circle at its first position once alpha hit 1.0). Use this corrected form:

```cpp
    POINT cp; GetCursorPos(&cp);
    RECT tr = frame(g_target);
    bool inside = PtInRect(&tr, cp) != 0;
    if (inside && g_circlePx > 0) {
        POINT nc = {cp.x - tr.left, cp.y - tr.top};
        if (nc.x != g_circleCenter.x || nc.y != g_circleCenter.y) {
            g_circleCenter = nc;
            g_lastMouseAt = GetTickCount();
            g_renderDirty = 1;
        }
    } else if (g_circleCenter.x >= 0) {
        g_circleCenter = {-1, -1}; g_renderDirty = 1;
    }
    // idle fade: alpha eases to 0 after 1500ms without cursor movement, back
    // to 1 on movement; ~300ms total at the 33ms poll rate (rate per tick)
    if (g_lastMouseAt) {
        DWORD idle = GetTickCount() - g_lastMouseAt;
        float targetA = (g_circlePx > 0 && inside && idle < 1500) ? 1.0f : 0.0f;
        float rate = (float)POLL_INTERVAL_MS / 300.0f;
        if (g_circleAlpha < targetA) { g_circleAlpha = fminf(1.0f, g_circleAlpha + rate); g_renderDirty = 1; }
        else if (g_circleAlpha > targetA) { g_circleAlpha = fmaxf(0.0f, g_circleAlpha - rate); g_renderDirty = 1; }
    }
```
(Fade = ~300ms at 33ms poll — the rate-based increment per poll tick.)

- [ ] **Step 3: Rewrite render_from_cache() with downscale/upscale + dual blur**:

```cpp
    for(int y=0;y<sh;y++)for(int x=0;x<sw;x++){

        unsigned r=0,g=0,b=0;
        for(int dy=0;dy<4;dy++)for(int dx=0;dx<4;dx++){
            const BYTE* p=&g_cache[((sy+dy)*w+(sx+dx))*4];
            r+=p[2]; g+=p[1]; b+=p[0];
        }
        auto& o=low[y*sw+x]; o.r=(BYTE)(r/16); o.g=(BYTE)(g/16); o.b=(BYTE)(b/16); o.a=255;
    }
    std::vector<BGRA> base=low, boost=low;
    int baseR = max(1,(int)(g_radius*g_curFocusF));           // Task 4 factor
    blur(base,sw,sh,baseR);
    int boostR = max(baseR+1, baseR*g_boostX100/100);
    if(g_circlePx>0 && g_circleAlpha>0.01f){ blur(boost,sw,sh,boostR); }
    else boost = base;
    std::vector<BGRA> out(w*h);
    int cr = (g_circleCenter.x<0)?-1:g_circleCenter.x, cc=(g_circleCenter.y<0)?-1:g_circleCenter.y;
    for(int y=0;y<h;y++)for(int x=0;x<w;x++){
        int ly=y/4, lx=x/4; if(ly>=sh)ly=sh-1; if(lx>=sw)lx=sw-1;   // clamp: w,h not multiple of 4
        const BGRA& lo = base[ly*sw+lx];
        if(cr<0){ out[y*w+x]=lo; continue; }
        float m = 0;
        if(g_circleAlpha>0.01f){
            int dx = x-cr, dy = y-cc;
            int d = (int)sqrtf((float)(dx*dx+dy*dy));
            m = circle_mask(d, g_circlePx, g_bandPx) * g_circleAlpha;
        }
        const BGRA& hi = boost[ly*sw+lx];
        out[y*w+x].r=(BYTE)(lo.r*(1-m)+hi.r*m);
        out[y*w+x].g=(BYTE)(lo.g*(1-m)+hi.g*m);
        out[y*w+x].b=(BYTE)(lo.b*(1-m)+hi.b*m);
        out[y*w+x].a=255;
    }
    apply_tint(out,(int)(g_tintPct*g_tintF));   // Task 4 fix round 1: g_tintF, NOT g_curFocusF*5/6
    render_layered(reinterpret_cast<const std::vector<BYTE>&>(out),w,h,(BYTE)(g_washAlpha*g_curFocusF));
}
```

- [ ] **Step 2: Add monitor pass** — in the autofrost monitor thread loop (existing 30s poll cadence is too slow for crash recovery — run this check every ~1s: add a cheap `Sleep(1000)` inner loop or reuse existing 800ms poll; plan decision: run inside the existing poll loop at its current cadence (800ms debug value) and note the v2.7 backlog item "poll 800ms→30s" applies to autofrost matching only — keep crash check at ~1s cadence with its own timer).

```cpp
void guard_overlays() {
    DWORD now = GetTickCount();
    std::vector<HWND> dead;
    EnterCriticalSection(&g_fxLock);
    for (auto& kv : g_overlays) {
        if (WaitForSingleObject(kv.second.hProcess, 0) == WAIT_OBJECT_0) dead.push_back(kv.first);
    }
    LeaveCriticalSection(&g_fxLock);
    for (HWND h : dead) {
        kill_overlay(h);   // erases map entries + closes handles
        if (g_shuttingDown || !IsWindow(h)) continue;
        auto& rc = g_restartCount[h];
        if (rc.second == 0) rc.second = now;
        if (rc.first >= 3 && now - rc.second < 30000) {
            tray_balloon(L"overlay crash loop", L"overlay crashed 3x in 30s — stopped");
            g_restartCount.erase(h);
            continue;
        }
        if (now - rc.second >= 30000) rc = {0, now};
        rc.first++;
        launch_overlay(h);
    }
}
```
